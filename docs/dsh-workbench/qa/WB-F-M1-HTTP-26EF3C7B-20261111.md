既有回归命令为 `python3 -m pytest -q tests/test_dsh_workbench_b_project_metadata_http_postgres.py tests/test_dsh_workbench_metadata_http_unready.py`。以下独立用例可存为临时文件，以仓库、tests 和 psycopg 环境加入 PYTHONPATH，在同一隔离 DSN 下用 `python3 -m pytest -q -p no:cacheprovider /tmp/f_m1_http_acl_exact.py::MetadataAcl` 运行。F 仅提交报告。

```python
import unittest
from test_dsh_workbench_b_project_metadata_http_postgres import ProjectMetadataHTTPPostgres as Fixture


class MetadataAcl(unittest.TestCase):
    def test_downgrade_revoke_restore_original_receipt(self):
        f = Fixture(methodName='test_native_rename_replays_original_receipt_and_retains_current_name_and_batch')
        self.addCleanup(f.doCleanups)
        f.setUp()
        f.db.store.create_batch('alice', 'P1', 'B1', '2027-01-01', 'fixture-batch')
        server = f.native_server()
        original = f.body('authorized rename', 'acl-committed')
        status, receipt = f.exchange(server, 'PATCH', original)
        self.assertEqual(status, 200)

        def snapshots():
            return {table: f.db.owner.execute('SELECT * FROM wb.' + table).fetchall()
                    for table in ('project_metadata_history','project_metadata_result',
                                  'project_metadata_outbox','request_log','audit','release_batch')}

        initial = snapshots()
        r1 = f.r1.read_bytes()
        denied = (403, {'error': {'code': 'FORBIDDEN'}})
        for sql, readable in [
            ("UPDATE wb.project_acl SET can_write=false WHERE project_id='P1' AND principal='alice'", True),
            ("DELETE FROM wb.project_acl WHERE project_id='P1' AND principal='alice'", False),
        ]:
            f.db.owner.execute(sql)
            status, body = f.exchange(server, 'GET')
            if readable:
                self.assertEqual((status, body['name'], body['can_rename']), (200, 'authorized rename', False))
                self.assertNotEqual(body['access_scope_revision'], receipt['access_scope_revision'])
            else:
                self.assertEqual((status, body), denied)
            self.assertEqual(f.exchange(server, 'PATCH', f.body('denied new rename', 'acl-denied', '2')), denied)
            self.assertEqual(f.exchange(server, 'PATCH', original), denied)
            self.assertEqual(snapshots(), initial)
            self.assertEqual(f.r1.read_bytes(), r1)

        f.db.owner.execute("INSERT INTO wb.project_acl VALUES ('P1','alice',true)")
        self.assertIs(f.exchange(server, 'GET')[1]['can_rename'], True)
        self.assertEqual(f.exchange(server, 'PATCH', original), (200, receipt))
        self.assertEqual(snapshots(), initial)
        self.assertEqual(f.r1.read_bytes(), r1)
```
